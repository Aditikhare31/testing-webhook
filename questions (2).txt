#!/bin/bash
set -e

# Update package index
apt-get update -y

# Install required packages to use apt over HTTPS
apt-get install -y \
    ca-certificates \
    curl \
    gnupg \
    lsb-release

# Create keyring directory for Docker
install -m 0755 -d /etc/apt/keyrings

# Add Docker’s official GPG key
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | \
    gpg --dearmor -o /etc/apt/keyrings/docker.gpg

chmod a+r /etc/apt/keyrings/docker.gpg

# Add the Docker apt repository
echo \
  "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] \
  https://download.docker.com/linux/ubuntu \
  $(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}") stable" \
  > /etc/apt/sources.list.d/docker.list

# Update apt package index again with the new repo
apt-get update -y

# Install Docker Engine and related packages
apt-get install -y \
    docker-ce \
    docker-ce-cli \
    containerd.io \
    docker-buildx-plugin \
    docker-compose-plugin

# Enable and start Docker
systemctl enable docker
systemctl start docker

# (Optional) Add default ubuntu user to docker group
usermod -aG docker ubuntu

# Show installed Docker version
docker --version

Step 1: Create a Simple App
A container that runs python HTTP server on port 80

mkdir health-demo  && cd health-demo

#Create Dockerfile
FROM python:3.10-slim

WORKDIR /app

#Simple web server
CMD python -m http.server 8080

# Health check
HEALTHCHECK --interval=10s --timeout=3s --retries=3 \
CMD curl -f http://localhost:8080 || exit 1
